function Test.Branches (score: i64, vip: bool) -> events {output{grade: str, bonus: i64}, error{message: str}}
  1 if truthy(sne(score, null))
  2   if present(vip)
  3     extra_value: i64 := add(score, 5)
  4     ran_extra: bool := true
  5   else
  6     ran_extra: bool := false
  7   if truthy(ge(score, 90))
  8     if ran_extra
  9       emit output {grade: "A", bonus: extra_value}
 10       ran_a: bool := true
 11     else
 12       ran_a: bool := false
 13   else
 14     emit output {grade: cond(truthy(ge(score, 50)), "pass", "fail"), bonus: 0}
 15 else
 16   emit error {message: "no score"}

CREATE OR REPLACE PACKAGE __sprig_l44_v1_lib AS
  PROCEDURE fail(code IN INT, msg IN VARCHAR);
END;

CREATE OR REPLACE PACKAGE BODY __sprig_l44_v1_lib AS
  PROCEDURE fail(code IN INT, msg IN VARCHAR) AS
    v VARCHAR(128) := LEFT(msg, 128);
  BEGIN
    SIGNAL SQLSTATE '45000' SET MYSQL_ERRNO = code, MESSAGE_TEXT = v;
  END;
END;

CREATE OR REPLACE PROCEDURE __sprig_l44_v1 (
    p_score IN BIGINT,
    p_vip IN BOOLEAN,
    p__event OUT VARCHAR(64),
    p__result OUT LONGTEXT
  )
  SQL SECURITY INVOKER
AS
  v_extra_value BIGINT;
  v_ran_extra BOOLEAN;
  v_ran_a BOOLEAN;
BEGIN
  IF (COALESCE((NOT (p_score <=> NULL)), 0) <> 0) THEN
    IF (COALESCE((COALESCE(p_vip, 0) <> 0), 0) <> 0) THEN
      v_extra_value := (p_score + 5);
      v_ran_extra := TRUE;
    ELSE
      v_ran_extra := FALSE;
    END IF;
    IF (COALESCE((p_score >= 90), 0) <> 0) THEN
      IF (COALESCE(v_ran_extra, 0) <> 0) THEN
        IF p__event IS NULL THEN
          p__event := 'output';
          p__result := JSON_OBJECT('grade', 'A', 'bonus', (v_extra_value + 0));
        END IF;
        RETURN;
        v_ran_a := TRUE;
      ELSE
        v_ran_a := FALSE;
      END IF;
    ELSE
      IF p__event IS NULL THEN
        p__event := 'output';
        p__result := JSON_OBJECT('grade', (CASE WHEN (COALESCE((p_score >= 50), 0) <> 0) THEN 'pass' ELSE 'fail' END), 'bonus', (0 + 0));
      END IF;
      RETURN;
    END IF;
  ELSE
    IF p__event IS NULL THEN
      p__event := 'error';
      p__result := JSON_OBJECT('message', 'no score');
    END IF;
  END IF;
  IF p__event IS NULL THEN
    __sprig_l44_v1_lib.fail(39866, 'No events raised');
  END IF;
END
