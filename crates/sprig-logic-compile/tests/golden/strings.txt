  2 en_result: bool := ends_with(name, "z")
  3 has_result: bool := contains(name, "a")
  4 len_result: i64 := len(name)
  5 low_result: str := lower(name)
  6 pos_result: i64 := index_of(name, "b")
  7 st_result: bool := starts_with(name, "Ab")
  8 up_result: str := upper(name)
  9 emit output {greeting: concat(concat(concat(concat(concat("Hi ", name), "! #"), num_str(n)), " "), num_str(div(n, 4))), len: len_result, upper: up_result, lower: low_result, hasA: has_result, startsAb: st_result, endsZ: en_result, posB: pos_result, joined: cat_value, same: seq(name, "abc"), before: lt(name, "b")}

CREATE OR REPLACE PACKAGE __sprig_l44_v1_lib AS
  PROCEDURE fail(code IN INT, msg IN VARCHAR);
  FUNCTION num_str(x IN DOUBLE) RETURN VARCHAR;
END;

CREATE OR REPLACE PACKAGE BODY __sprig_l44_v1_lib AS
  PROCEDURE fail(code IN INT, msg IN VARCHAR) AS
    v VARCHAR(128) := LEFT(msg, 128);
  BEGIN
    SIGNAL SQLSTATE '45000' SET MYSQL_ERRNO = code, MESSAGE_TEXT = v;
  END;
  FUNCTION num_str(x IN DOUBLE) RETURN VARCHAR AS
  BEGIN
    RETURN CASE WHEN x IS NULL THEN NULL
                WHEN x = FLOOR(x) AND ABS(x) < 9007199254740992
                THEN CAST(CAST(x AS SIGNED) AS CHAR) ELSE CAST(x AS CHAR) END;
  END;
END;
